<script type="module">
// ──────────────────────────────────────────────
// FIREBASE設定（あなた専用の鍵）
// ──────────────────────────────────────────────
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-app.js";
import { getFirestore, collection, addDoc, query, orderBy, onSnapshot, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-firestore.js";

const firebaseConfig = {
  apiKey: "AIzaSyBeYO4jT79o7WUAbuWSuNPDU1X084H7O-w",
  authDomain: "hearme-40eb9.firebaseapp.com",
  projectId: "hearme-40eb9",
  storageBucket: "hearme-40eb9.firebasestorage.app",
  messagingSenderId: "291597042418",
  appId: "1:291597042418:web:c08c163c94ea1f77cec66b"
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

// ──────────────────────────────────────────────
// アプリロジック
// ──────────────────────────────────────────────

// 画面遷移
window.go = (id) => {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('on'));
    document.getElementById('p-' + id).classList.add('on');
};

// ユーザー名保存
window.saveUserName = () => {
    const name = document.getElementById('user-name-input').value;
    if(!name) return alert('名前を入れてね');
    localStorage.setItem('hm_user', name);
    go('home');
};

// 投稿機能（クラウド保存）
window.doPost = async () => {
    const song = document.getElementById('song-in').value;
    const artist = document.getElementById('artist-in').value;
    const comment = document.getElementById('cmt-in').value;
    const name = localStorage.getItem('hm_user');

    if(!song || !artist) return alert('曲とアーティストを教えて！');

    try {
        await addDoc(collection(db, "posts"), {
            song, artist, comment, name,
            time: serverTimestamp()
        });
        localStorage.setItem('last_post_date', new Date().toDateString());
        go('home');
        // 入力クリア
        document.getElementById('song-in').value = '';
        document.getElementById('artist-in').value = '';
        document.getElementById('cmt-in').value = '';
    } catch (e) {
        alert('エラーが発生しました。Firebaseの設定を確認してください。');
    }
};

// リアルタイム読み込み（BeReal風制限付き）
function listenPosts() {
    const q = query(collection(db, "posts"), orderBy("time", "desc"));
    const hasPostedToday = localStorage.getItem('last_post_date') === new Date().toDateString();

    onSnapshot(q, (snapshot) => {
        const feed = document.getElementById('feed');
        feed.innerHTML = '';
        
        snapshot.forEach((doc) => {
            const data = doc.data();
            const card = document.createElement('div');
            card.className = 'card' + (hasPostedToday ? '' : ' blur');
            
            // 時間の表示変換
            let timeStr = 'Just now';
            if(data.time) {
                const date = data.time.toDate();
                timeStr = date.getHours() + ':' + String(date.getMinutes()).padStart(2, '0');
            }

            card.innerHTML = `
                <div style="display:flex; justify-content:space-between; font-size:12px; color:var(--tx2); margin-bottom:8px;">
                    <span>${data.name || '匿名'}</span>
                    <span>${timeStr}</span>
                </div>
                <div style="font-weight:600; font-size:18px;">${data.song}</div>
                <div style="color:var(--accent); font-size:14px; margin-bottom:10px;">${data.artist}</div>
                <div style="font-size:14px; line-height:1.4;">${data.comment || ''}</div>
            `;
            feed.appendChild(card);
        });

        if(!hasPostedToday) {
            const msg = document.createElement('div');
            msg.style.cssText = "text-align:center; padding:20px; color:var(--accent); font-weight:600;";
            msg.innerHTML = "✨ 今日の曲を投稿して、友達の曲をチェックしよう！";
            feed.prepend(msg);
        }
    });
}

// 起動時チェック
if(localStorage.getItem('hm_user')) {
    go('home');
} else {
    go('login');
}
listenPosts();
