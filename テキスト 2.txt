      await fetch('/api/register-token', {

<script>
// ── State ──
let _song=null, _emo=null, _mdPlay=false, _mdVid='', _brInt=null, _brSec=180, _fromBr=false;
const YTK = 'AIzaSyD_placeholder_replace_me';

// ── Boot ──
function boot() {
  go('home');
  if ('serviceWorker' in navigator) {
    navigator.serviceWorker.register('sw.js').then(reg => {
      navigator.serviceWorker.addEventListener('message', e => {
        if (e.data?.type === 'BEREAL_TRIGGER') triggerBereal();
      });
    }).catch(() => {});
  }
  if ('Notification' in window && Notification.permission === 'granted') {
    document.getElementById('notif-bar').style.display = 'none';
  }
  // Check URL param
  if (location.search.includes('bereal=1')) triggerBereal();
}

// ── Navigation ──
function go(id) {
  document.querySelectorAll('.scr').forEach(s => s.classList.remove('on'));
  document.getElementById('scr-' + id)?.classList.add('on');
}

// ── Match tabs ──
function mTab(t) {
  document.getElementById('mt-s').classList.toggle('on', t === 's');
  document.getElementById('mt-f').classList.toggle('on', t === 'f');
  document.getElementById('mp-s').style.display = t === 's' ? 'block' : 'none';
  document.getElementById('mp-f').style.display = t === 'f' ? 'block' : 'none';
}

// ── Notification ──
async function reqNotif() {
  if (!('Notification' in window)) { toast('このブラウザは通知未対応です'); return; }
  const p = await Notification.requestPermission();
  if (p === 'granted') {
    document.getElementById('notif-bar').style.display = 'none';
    toast('🔔 通知をONにしました！');
    const token = await window._getFCMToken?.();
    if (token) console.log('FCM registered');
  } else {
    toast('設定から通知を許可してください');
  }
}

// ── BeReal ──
window.triggerBereal = function() {
  _brSec = 180;
  document.getElementById('bereal-ov').classList.add('show');
  updBr();
  _brInt = setInterval(() => {
    _brSec--;
    updBr();
    if (_brSec <= 0) {
      clearInterval(_brInt);
      document.getElementById('bereal-ov').classList.remove('show');
      toast('時間切れ…遅れた投稿として残ります');
    }
  }, 1000);
  // Also push browser notification if in background
  if ('Notification' in window && Notification.permission === 'granted') {
    try { new Notification('hearme 🎵', { body: '今の気持ちの曲は？', icon: 'icons/icon-192.png', tag: 'hm' }); } catch(e){}
  }
};
function updBr() {
  const m = String(Math.floor(_brSec / 60)).padStart(2, '0');
  const s = String(_brSec % 60).padStart(2, '0');
  document.getElementById('br-t').textContent = `${m}:${s}`;
}
function openPostBereal() {
  clearInterval(_brInt);
  document.getElementById('bereal-ov').classList.remove('show');
  _fromBr = true;
  document.getElementById('pt-br').style.display = 'flex';
  go('post');
}
function skipBereal() {
  clearInterval(_brInt);
  document.getElementById('bereal-ov').classList.remove('show');
  toast('遅れた投稿として記録されます');
}

}, 10000);

// ── Match detail ──
function openMD(d) {
  document.getElementById('md-a').textContent = d.a;
  document.getElementById('md-n').textContent = d.n;
  document.getElementById('md-e').textContent = d.e;
  document.getElementById('md-d').textContent = d.d;
  document.getElementById('md-s').textContent = d.s;
  document.getElementById('md-ar').textContent = d.ar;
  document.getElementById('md-c').textContent = `「${d.c}」`;
  _mdVid = d.v || '';
  const th = document.getElementById('md-th');
  th.innerHTML = d.th
    ? `<img src="${d.th}" style="width:100%;height:100%;object-fit:cover;" onerror="this.remove()"/>`
    : `<svg width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.2" opacity=".25"><path d="M9 18V5l12-2v13"/><circle cx="6" cy="18" r="3"/><circle cx="18" cy="16" r="3"/></svg>`;
  _mdPlay = false;
  document.getElementById('md-pi').classList.remove('v');
  document.getElementById('md-ico').style.display = 'block';
  document.getElementById('md-lbl').textContent = 'プレビューを再生';
  document.querySelectorAll('.rb').forEach(b => b.classList.remove('on'));
  document.getElementById('md-bg').classList.add('on');
}
function closeMD(e) {
  if (e.target === document.getElementById('md-bg')) document.getElementById('md-bg').classList.remove('on');
}
function mdPlay() {
  _mdPlay = !_mdPlay;
  document.getElementById('md-pi').classList.toggle('v', _mdPlay);
  document.getElementById('md-ico').style.display = _mdPlay ? 'none' : 'block';
  document.getElementById('md-lbl').textContent = _mdPlay ? '再生中...' : 'プレビューを再生';
  if (_mdPlay && _mdVid) window.open(`https://www.youtube.com/watch?v=${_mdVid}`, '_blank');
}
function doR(btn, emoji) {
  document.querySelectorAll('.rb').forEach(b => b.classList.remove('on'));
  btn.classList.add('on');
  toast(`${emoji} リアクションを送りました`);
}

// ── Events ──
function evJoin(btn) {
  const j = btn.classList.contains('joined');
  btn.classList.toggle('joined', !j);
  btn.textContent = j ? '参加する' : '参加済み ✓';
  if (!j) toast('イベントに参加しました 🎉');
}
function evChip(c) {
  document.querySelectorAll('.ev-chip').forEach(x => x.classList.remove('on'));
  c.classList.add('on');
}

// ── Post / YouTube search ──
async function doSearch() {
  const q = document.getElementById('srch-in').value.trim();
  if (!q) return;
  document.getElementById('srch-load').style.display = 'flex';
  document.getElementById('srch-res').innerHTML = '';
  try {
    const r = await fetch(`https://www.googleapis.com/youtube/v3/search?part=snippet&q=${encodeURIComponent(q + ' 音楽 official')}&type=video&videoCategoryId=10&maxResults=5&key=${YTK}`);
    const d = await r.json();
    document.getElementById('srch-load').style.display = 'none';
    if (d.error || !d.items?.length) { dummy(q); return; }
    d.items.forEach(i => addR(i.id.videoId, i.snippet.title, i.snippet.channelTitle, i.snippet.thumbnails.medium?.url));
  } catch { document.getElementById('srch-load').style.display = 'none'; dummy(q); }
}
function dummy(q) {
  [['x8VYWazR5mE', `${q} - Official MV`, 'Music Channel'], ['SX_ViT4Ra7k', `${q} (Live)`, 'Artist Official'], ['gdZLi9oWNZg', `${q} - Lyric Video`, 'Vevo']].forEach(([v, t, c]) => addR(v, t, c, `https://img.youtube.com/vi/${v}/mqdefault.jpg`));
}
function addR(vid, title, ch, thumb) {
  const d = document.createElement('div');
  d.className = 'res-item';
  d.innerHTML = `<div class="res-th"><img src="${thumb||''}" onerror="this.style.display='none'" alt=""/></div><div class="res-info"><div class="res-ttl">${esc(title)}</div><div class="res-ch">${esc(ch)}</div></div><div class="res-sel" id="rs${vid}"><svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><polyline points="20 6 9 17 4 12"/></svg></div>`;
  d.onclick = () => pickS(vid, title, ch, thumb);
  document.getElementById('srch-res').appendChild(d);
}
function pickS(vid, title, ch, thumb) {
  _song = { vid, title, ch, thumb };
  document.getElementById('sel-img').src = thumb || '';
  document.getElementById('sel-t').textContent = title;
  document.getElementById('sel-a').textContent = ch;
  document.getElementById('sel-prev').style.display = 'flex';
  document.querySelectorAll('.res-sel').forEach(e => e.classList.remove('on'));
  document.getElementById('rs' + vid)?.classList.add('on');
  updPt();
}
function pickE(c) {
  document.querySelectorAll('.ec').forEach(x => x.classList.remove('on'));
  c.classList.add('on');
  _emo = c.textContent;
  updPt();
}
function updPt() { document.getElementById('pt-go').classList.toggle('ok', !!(_song && _emo)); }
function updC() { document.getElementById('cmt-c').textContent = document.getElementById('cmt-ta').value.length; }
function doPost() {
  if (!_song || !_emo) { toast('曲と気分を選んでください'); return; }
  const posts = JSON.parse(localStorage.getItem('hm_posts') || '[]');
  posts.unshift({ song: _song.title, artist: _song.ch, vid: _song.vid, thumb: _song.thumb, emo: _emo, comment: document.getElementById('cmt-ta').value, time: new Date().toISOString(), fromBr: _fromBr });
  localStorage.setItem('hm_posts', JSON.stringify(posts.slice(0, 50)));
  go('home');
  toast(_fromBr ? '投稿しました！みんなに届けました 🎵🔴' : '投稿しました 🎵');
  _song = null; _emo = null; _fromBr = false;
  document.getElementById('srch-in').value = '';
  document.getElementById('srch-res').innerHTML = '';
  document.getElementById('sel-prev').style.display = 'none';
  document.getElementById('cmt-ta').value = '';
  document.getElementById('cmt-c').textContent = '0';
  document.querySelectorAll('.ec').forEach(c => c.classList.remove('on'));
  document.getElementById('pt-go').classList.remove('ok');
  document.getElementById('pt-br').style.display = 'none';
}

// ── Toast ──
let _tt = null;
function toast(msg) {
  const t = document.getElementById('toast-el');
  t.textContent = msg;
  t.classList.add('show');
  clearTimeout(_tt);
  _tt = setTimeout(() => t.classList.remove('show'), 2800);
}

function esc(s) { return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
